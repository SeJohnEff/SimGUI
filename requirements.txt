# SimGUI dependencies
#
# tkinter is part of the Python standard library (no pip install needed).
# On Debian/Ubuntu you may need: sudo apt install python3-tk
#
# PyQt6 is required for the Qt-based UI (main.py).
PyQt6>=6.6        # Qt6 bindings for the new UI
Pillow>=10.0      # Image support for tkinter (required for PNG icons in PyInstaller bundle)
#
# The following are only needed for card-reader hardware features.
# They can be omitted for CSV-only / offline usage.
pyscard>=2.0.0    # PC/SC smart-card interface (card reader communication)
pytlv>=0.3        # TLV parsing for SIM card data structures
#
# REQUIRED for gialersim key verification (not optional). The offline USIM
# AUTHENTICATE self-check (managers/gialersim_selfcheck.py) computes Milenage to
# prove Ki/OPc committed; without a crypto backend that check cannot run, and
# fail-closed policy then refuses to report the card as programmed. Provides the
# `Crypto` package (the self-check also accepts `Cryptodome`/pycryptodomex).
pycryptodome>=3.19  # Milenage AES for the gialersim AUTHENTICATE self-check

# Development / testing
# pytest>=7.0
pytest-qt>=4.4  # Qt widget testing
pytest-env>=0.8.2  # Environment variable support for pytest
